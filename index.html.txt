<script>
const tags=document.querySelectorAll('.tag');
const cards=document.querySelectorAll('.video-card');
const search=document.getElementById('search');
tags.forEach(t=>{
    t.onclick=()=>{
        tags.forEach(i=>i.classList.remove('active'));
        t.classList.add('active');
        let type=t.dataset.type;
        cards.forEach(c=>{
            if(type==='all'||c.dataset.type===type) c.style.display='block';
            else c.style.display='none';
        })
    }
})
search.oninput=()=>{
    let kw=search.value.toLowerCase();
    cards.forEach(c=>{
        let name=c.querySelector('.title').innerText.toLowerCase();
        c.style.display=name.includes(kw)?'block':'none';
    })
}
const modal=document.getElementById('modal');
const playVideo=document.getElementById('playVideo');
function openVideo(el){
    playVideo.src=el.querySelector('video').src;
    modal.style.display='flex';
    playVideo.play();
}
function closeVideo(){
    modal.style.display='none';
    playVideo.pause();
}
modal.onclick=e=>{if(e.target===modal) closeVideo()}
</script>
